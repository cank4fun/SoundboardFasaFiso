# DİL / LANGUAGE

language=en

# SES ÇIKIŞLARI
# output ana çıkıştır. "default" Windows varsayılanını kullanır.

# monitor kendi duyacağın ikinci çıkıştır. İlk açılışta kapalıdır.
# "default" Windows varsayılanını kullanır; "none" monitörü kapatır.

# MİKROFON MİKSERİ
# Mikrofonu ana çıkışa ve/veya monitöre gerçek zamanlı ekler.

# MİKROFON İŞLEME
# Ana anahtar varsayılan olarak kapalıdır; eski config davranışı korunur.
# Noise suppression ve AGC altyapısı tamamlanana kadar kapalı tutulur.

microphone_processing_preset=natural
microphone_high_pass_enabled=true
microphone_high_pass_hz=80.0
microphone_noise_suppression_enabled=false

# AUDIO GECİKME AYARLARI
# 0 yazılırsa cihazın/Windows'un varsayılanı kullanılır.

# UYGULAMA
# Windows ile otomatik başlatma kullanıcı hesabına özel kayıt defteri ayarı kullanır.

# Eski config uyumluluğu için tutulur. GUI sürümü konsolsuz açılır.
show_console_on_start=false
check_updates_on_start=true

# KONTROLLER
stop=F11

# TUŞ ATAMALARI
# Varsayılan: volume=1.00 ve mode=restart
F1=example.wav|volume=0.80|mode=restart
F2=example2.wav|volume=1.00|mode=toggle
F3=example.wav|volume=0.60|mode=loop
F4=example2.wav|volume=1.00|mode=overlap
F5=example.mp3|volume=0.80|mode=restart
F6=example.flac|volume=1.00|mode=overlap

# MODLAR
# restart = Her basışta sesi baştan çalar.
# toggle  = Bir basışta çalar, tekrar basışta durdurur.
# loop    = Döngüye alır, tekrar basışta durdurur.
# overlap = Her basışta yeni voice açar; en fazla 8 kopya aynı anda çalar.
